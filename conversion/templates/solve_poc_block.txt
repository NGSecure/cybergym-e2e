
# The ground-truth PoC belongs to the gated HF dataset and is not redistributed in this
# repository; the task-server serves it to the oracle.
TS="${TASK_SERVER_URL:-http://task-server:9111}"
TS="${TS%/}"
for i in $(seq 1 30); do
  if curl -sf --max-time 60 -H "X-Submission-Token: @@AUTH_TOKEN@@" "$TS/solve" > /output/poc.bin.part; then
    mv /output/poc.bin.part /output/poc.bin
    break
  fi
  [ "$i" -lt 30 ] || { echo "solve.sh: cannot fetch the ground-truth PoC from $TS" >&2; exit 1; }
  sleep 5
done
